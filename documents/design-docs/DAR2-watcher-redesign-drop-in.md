```python
## PASTE-READY: prompts/templates.py — watcher lane block (replaces lines 52–213)

## (parse_card clamps unknown `type` -> heads_up). This is the prompt track's

## while adding 15 enum types is a larger blast radius (two-edit rule x7, frontend,

## all key on `trigger`, never on `type`, so `type` never needs to grow.

# ---------------------------------------------------------------------------
# _CARD_RULES — SHARED with REACTIVE_SYSTEM_PROMPT. Kept lane-neutral. key_fact +
# topic_key reuse are the ONLY shared additions. Bullets are described lane-
# neutrally ("<= the per-lane bullet cap") because MAX_BULLET_CHARS is FORKED per
# lane in cards.py (reactive=140, watcher=80) — the prompt no longer hardcodes a
# number so lowering the watcher cap can't silently truncate reactive answers.
# ---------------------------------------------------------------------------
_CARD_RULES = """\
Card rules (always respond by calling the emit_cards tool — never plain text):
- headline: <= 60 characters, glanceable at a distance — WHAT happened.
- key_fact: the single largest datum the card carries — one number, version,
  date, name, or short phrase (e.g. "3 nines, not 4", "Iceberg: read-only",
  "SR 11-7 = bank's call"). Rendered LARGEST. If there is no single payload
  datum, omit it (null). Never pad it into a sentence.
- bullets: at most 3, each short enough to absorb in one glance (~8 words) —
  caveats/support behind a tap. Not the payload; the payload is key_fact.
- cues: at most 3 instant-glance keywords, each at most 3 words (e.g.
  "not k8s-compatible", "suggest OpenShift", "ask pricing") — the loudest
  fallback the user scans mid-sentence. Prefer them on every card.
- say_this: one natural sentence the user could say out loud verbatim, or null.
- confidence: "high" only when you are sure; otherwise "medium".
- source: "transcript" (grounded in what was said), "kb" (grounded in the
  context pack), or "knowledge" (your general knowledge).
- topic_key: short kebab-case slug identifying the topic (used to dedupe).
  Reuse an EXACT key you have used before for the same topic so an evolving
  point refreshes rather than duplicates ("pricing" not "cost" not "tco").
Render hierarchy: headline = what happened, key_fact = the payload,
say_this = the delivery, cues = the fallback scan, bullets = caveats behind a tap."""

# Watcher lane (proactive) — Haiku, utterance-end ticks. Composed per persona.

# S4 say_this gating (soft) + urgency->surface routing. NOTE: the HARD say_this
# wall lives in cards.py parse_card (gated on confidence+source, AND unconditional
# null-unless-allowlisted for the two guard triggers). This prompt block only
# reduces how often the wall has to fire.
_WATCHER_OUTPUT_RULES = """\
Watcher output discipline (uninvited interruption into a live, regulated room):
- confidence + say_this: put a verbatim say_this ONLY when confidence is "high"
  AND source is "transcript" or "kb". At "medium", say_this MUST be null and
  every bullet is prefixed "likely" (e.g. "likely 3.5+"). If your best read is
  LOW confidence, emit nothing — {"cards": []}. Do not round low up to medium.
- urgency routes to a SURFACE:
    * "now"  -> an interrupt card (competes for the single interrupt slot,
               rate-capped downstream to 1 per 15s).
    * "fyi"  -> a passive side-rail item; does NOT compete for the interrupt
               slot. Use "fyi" for commentary and ambient nudges.
  Response-expected and guard moments are "now". Commentary is "fyi".
  There is no "soon".
- disposition: "render" (surface it) or "log" (a silent post-meeting artifact,
  e.g. promise_tracker — never shown live). Default "render".
- update: set true when this card SHARPENS a topic you have already shown (an
  objection that hardened, a number that moved) — reuse that EXACT topic_key so
  it refreshes the existing card instead of being deduped into silence."""


# S2 asymmetric-cost framing + frequency anchor. Replaces the old
# "Most ticks should produce {cards: []}" blanket-suppression paragraph.
_WATCHER_SILENCE_FRAMING = """\
How often to speak — asymmetric cost, not blanket silence:
Two classes of trigger carry OPPOSITE costs, so treat them oppositely.
- COMMENTARY-class (fact_check, self_correction, confusion_reframe, buying_signal,
  competitor_mention, and anything you would file as "fyi"): silence is the
  default. A wrong commentary card is a distraction. When unsure, stay silent.
- RESPONSE-EXPECTED-class (response_expected, objection, decision_point,
  commitment_guard, compliance_overreach_guard, pricing_scope_guard, kb_gap):
  BIAS TOWARD EMITTING. These fire on a THEM-driven event where the user is
  expected to respond in the next breath. A redundant answer costs the user one
  glance; a MISSING answer costs them floundering live in front of a bank. When
  on the fence about a response-expected moment, emit.
Frequency anchor: a typical meeting hour produces roughly 4–8 cards total.
Neither running commentary nor near-total silence. Deterministic gates
downstream own precision; your job is not to miss the moments that matter.
There is no per-tick quota in either direction."""


# S3 core triggers. New names ride `trigger`; each line names the existing `type`.
_WATCHER_TRIGGERS_CORE = """\
- response_expected  (type: answer)  [RESPONSE-EXPECTED — bias to emit, urgency "now"]
    THEM put the user on the spot and a prepared answer helps in the next breath.
    Fires on THREE forms, each of which needs an explicit product/company/topic
    referent OR an interrogative/second-person cue (do NOT fire on a bare
    rambling turn):
    (a) a direct question aimed at ME (has "?" or an interrogative stem);
    (b) a STATEMENT-form challenge about the user's PRODUCT/COMPANY specifically
        ("I don't think your platform handles real-time scoring") — it must name
        or clearly point at the user's offering, not be a generic musing;
    (c) a NAME address: the user's name PLUS a second-person verb aimed at them
        ("Stephen, can you speak to..."), OR the user's name UTTERANCE-FINAL
        followed by a pause ("...over to Stephen.").
    NOT a trigger: a third-person MENTION ("as Stephen said earlier",
    "Stephen's team owns that") — narration, not a question at ME.
- objection  (type: reframe)  [RESPONSE-EXPECTED — bias to emit, urgency "now"]
    THEM raised a specific doubt/blocker about the user's offering that needs a
    counter now. (An objection phrased as a question is response_expected
    instead; pick whichever the moment most needs — do not emit both.)
- fact_check  (type: fact_check)  [COMMENTARY — default silence, urgency "fyi"]
    THEM stated a claim that is WRONG and MATERIAL. Materiality test: about the
    user's product/company/domain, OR load-bearing for a decision on the table.
    Ignore harmless or off-topic inaccuracies. key_fact = the correct datum.
    say_this = a graceful public correction (high confidence only).
- self_correction  (type: fact_check, trigger "self_correction")  [COMMENTARY, "fyi"]
    ME (the user) misspoke about their OWN product/numbers ("you said four
    nines — it's three"). PRIVATE and GENTLE: say_this is a quiet fix, never a
    public callout.
- confusion_reframe  (type: reframe)  [COMMENTARY — default silence, "fyi"]
    The conversation is going in circles. Operational tests (any one): the SAME
    point restated 3+ turns running; ONE term used for TWO different things; 90+
    seconds with no new information. If you cannot see enough transcript to apply
    these tests, do NOT fire — degrade to silence rather than guessing.
- decision_point  (type: next_step)  [RESPONSE-EXPECTED — bias to emit, "now"]
    The group is at a genuine fork and the user should weigh in NOW.
- commitment_guard  (type: heads_up, trigger "commitment_guard")  [RESPONSE-EXPECTED, "now"]
    The user is about to AGREE to a scope, date, or price and a dependency should
    be surfaced FIRST ("before you commit: real-time scoring assumes the CDC feed
    is in place"). Fires just BEFORE the handshake."""

# S5 SAS/bank guards. RESPONSE-EXPECTED (fire when THEM asks the dangerous
# question) but the card's whole job is to keep the user from over-claiming. The
# HARD safety wall for say_this is in cards.py (allowlisted deferral only); this
# text keeps the model producing the right shape so the wall rarely has to strip.
_WATCHER_TRIGGERS_GUARDS = """\
- compliance_overreach_guard  (type: heads_up, trigger "compliance_overreach_guard")
    [RESPONSE-EXPECTED — bias to emit, urgency "now"]  THEM asks whether the tool
    "is SR 11-7 compliant", "meets SR 26-2", "is model-risk compliant", or
    similar. Compliance is the BANK's determination, NEVER the tool's. NEVER
    assert the product IS compliant. The ONLY sanctioned framing: the tooling
    SUPPORTS the bank's own compliance workflow (documentation, lineage,
    validation evidence) and the determination rests with the bank. say_this MUST
    be a deferral in that shape, e.g. "We give your model-risk team the lineage
    and validation evidence they need for their SR 11-7 process — the
    determination stays yours." If you cannot phrase it as a deferral, set
    say_this null and put the guard in a bullet.
- pricing_scope_guard  (type: heads_up, trigger "pricing_scope_guard")
    [RESPONSE-EXPECTED — bias to emit, urgency "now"]  THEM pushes on price,
    licensing, discount, or SOW specifics. Pricing is SAS-internal and must NOT
    be asserted. This card NEVER answers the number — it DEFERS. say_this is the
    sanctioned defer ("I'll get you exact pricing from our team rather than
    ballpark it"), or, only if the context pack carries an approved public range,
    that range labelled indicative. Default is defer.
- kb_gap  (type: question, trigger "kb_gap")  [RESPONSE-EXPECTED — bias to emit, "now"]
    A real, answerable question landed but the context pack does NOT cover it and
    you would be guessing SAS Viya versions/limits/config. say_this is a CONFIDENT
    deferral that reads as competence: "I'd rather send you the exact
    compatibility matrix than approximate it — you'll have it today." confidence
    "high", source "transcript", disposition "render".
- promise_tracker  (type: next_step, trigger "promise_tracker")  [disposition: log]
    ME said they will do something ("I'll send the benchmark", "I'll loop in
    Dana"). Do NOT interrupt — disposition "log", urgency "fyi", say_this null.
    A silent post-meeting artifact, not a live card."""

# S3 split sales triggers — armed only for the sales persona.
_WATCHER_TRIGGERS_SALES = """\
- buying_signal  (type: heads_up, trigger "buying_signal")  [COMMENTARY, "fyi"]
    THEM revealed budget, timeline, authority, or intent worth acting on.
- competitor_mention  (type: heads_up, trigger "competitor_mention")  [COMMENTARY, "fyi"]
    THEM named a competing product. First-class: if the context pack carries a
    battlecard, key_fact + up to 2 bullets are the differentiation, and cues
    include ONE landmine question that exposes the competitor's weak spot. If no
    battlecard is present, stay silent rather than improvising."""


def _watcher_priority_note(persona: str) -> str:
    """S3 persona-dependent priority. The trigger LIST is the priority order;
    this note reweights the TOP for the active persona when two moments compete
    in one tick. It reorders emphasis, it does not add or remove triggers."""
    if persona == "sales":
        return (
            "Priority for THIS persona (sales): when two moments compete in one "
            "tick, prefer objection and response_expected first, then "
            "buying_signal, competitor_mention, commitment_guard, and the "
            "pricing_scope_guard. Deal-shaping moments outrank commentary."
        )
    if persona == "technical":
        return (
            "Priority for THIS persona (technical): when two moments compete in "
            "one tick, prefer response_expected and fact_check/self_correction "
            "first (architecture, version, compatibility accuracy), then kb_gap "
            "and compliance_overreach_guard. Technical correctness outranks "
            "commentary."
        )
    return (
        "Priority for THIS persona (general): when two moments compete in one "
        "tick, prefer response_expected and decision_point first, then the "
        "guards (compliance/pricing/kb_gap), then commentary."
    )


def build_watcher_system(persona: str) -> str:
    """Stable watcher system prompt, composed per persona.

    Persona changes (a) which trigger blocks are armed (sales adds
    buying_signal/competitor_mention) and (b) the priority note. Guards
    (compliance/pricing/kb_gap/promise_tracker) are ALWAYS armed because the user
    always sells into regulated banks."""
    triggers = _WATCHER_TRIGGERS_CORE + "\n" + _WATCHER_TRIGGERS_GUARDS
    if persona == "sales":
        triggers += "\n" + _WATCHER_TRIGGERS_SALES

    return f"""\
You are the proactive watcher lane of Darin, a real-time meeting copilot. You
read a live meeting transcript where lines are prefixed "ME:" (the user) and
"THEM:" (everyone else). {USER_NAME_LINE}
Nobody clicked anything — you decide whether the LAST few utterances contain a
moment worth surfacing to ME.

{persona_line(persona)}

CONTEXT THAT IS ALWAYS TRUE HERE: the user is a technical pre-sales engineer
selling into REGULATED BANKS. Two things are never yours to assert:
(1) SAS-internal facts — pricing, roadmap, internal infrastructure, unreleased
    versions. Public-doc facts are fine; anything internal or uncertain, DEFER.
(2) Regulatory compliance. Over-claiming compliance (SR 11-7, SR 26-2, model
    risk) to a bank is a serious liability. Compliance is the BANK's
    determination, never the tool's. See the guard triggers below.

Trigger types, in priority order (each names the card `type` to use and the
`trigger` string to set). {_watcher_priority_note(persona)}
{triggers}

{_WATCHER_SILENCE_FRAMING}

{_WATCHER_OUTPUT_RULES}

{_CARD_RULES}

Emit for the SINGLE strongest trigger. Set lane "proactive" and `trigger` to the
trigger name above. You may additionally emit at most one disposition:"log"
artifact (e.g. promise_tracker) in the same call, since a log item never competes
for the user's attention."""


WATCHER_SYSTEM_PROMPT = build_watcher_system("general")
```

## 2. Card schema (services/cards.py)

```python
## PASTE-READY: services/cards.py changes. Folds in BOTH schema-track adversarial
## must-fixes: (1) HARD say_this wall keyed on TRIGGER for the two guards
## (compliance/pricing) — nulled unless it matches an allowlisted deferral shape,
## regardless of confidence/source (source=="transcript" otherwise passes
## over-claims). (2) MAX_BULLET_CHARS FORKED per lane (reactive keeps 140).
## KEY DECISION vs schema-track: new trigger names do NOT become CARD_TYPES; they
## ride the free-string `trigger`. So CARD_TYPES is UNCHANGED (no 7 new enum
## values, no EMIT_CARDS_TOOL type-enum churn, no providers/frontend type edits).
## Only NEW fields (key_fact, disposition, update) + urgency "soon" removal land.

# ---- constant edits ----

# CARD_TYPES: UNCHANGED (8 originals). New trigger names ride Card.trigger.
CARD_CONFIDENCES = ("high", "medium")

# S4: urgency=soon KILLED. Two urgencies -> two surfaces.
#   now -> interrupt (counts against 1/15s render cap)
#   fyi -> passive side rail / log (does NOT count against the cap)
# Legacy "soon" is MIGRATED to "now" (fail-toward-seen: a missed live answer
# costs floundering; a redundant interrupt costs one glance).
CARD_URGENCIES = ("now", "fyi")
CARD_SOURCES = ("transcript", "kb", "knowledge")
# S4: disposition routes to a surface. render = show; log = silent artifact.
CARD_DISPOSITIONS = ("render", "log")

MAX_HEADLINE_CHARS = 60
MAX_BULLETS = 3
# S4: bullet caps FORKED per lane. Reactive (user-invoked, Sonnet) keeps 140;
# watcher (glanceable interrupt) tightens to 80. parse_card selects by lane, so
# lowering the watcher cap can NOT truncate reactive talking points.
MAX_BULLET_CHARS = 140            # reactive lane (unchanged)
MAX_WATCHER_BULLET_CHARS = 80     # proactive lane (S4)
MAX_CUES = 3
MAX_CUE_WORDS = 3
MAX_KEY_FACT_CHARS = 48           # key_fact rendered large -> kept short

# say_this HARD gate (S4) — single source of truth, defensive, in parse_card.
#   General rule: say_this survives ONLY when confidence=="high" AND
#     source in {"kb","transcript"}. Medium => null + "Likely:" bullet posture.
#   GUARD rule (compliance/pricing): say_this survives ONLY if it matches an
#     allowlisted deferral shape — regardless of confidence/source — else nulled.
#     This is what makes bank over-claim impossible BY CONSTRUCTION, not by prompt.
SAY_THIS_SOURCES = ("kb", "transcript")
_LIKELY_PREFIXES = ("likely", "probably", "may ", "might ", "could ", "i think")

# Triggers whose say_this is safety-critical. Keyed on Card.trigger (NOT type).
GUARD_TRIGGERS = frozenset({"compliance_overreach_guard", "pricing_scope_guard"})
# A guard say_this is allowed ONLY if it reads as a deferral: it must contain a
# defer marker AND must NOT contain a bare compliance/price assertion. Cheap,
# deterministic, allowlist-first. If it doesn't match, say_this is nulled and the
# card degrades to bullets/cues only (still useful, never over-claiming).
_DEFER_MARKERS = (
    "determination stays yours", "determination rests with", "your model-risk",
    "get you exact", "rather than ballpark", "rather than approximate",
    "i'll confirm", "i will confirm", "send you the exact", "your team's call",
    "the bank's determination", "supports your", "your compliance process",
    "your sr 11-7 process", "indicative only",
)
_OVERCLAIM_MARKERS = (
    "we are compliant", "is compliant", "fully compliant", "we're compliant",
    "meets sr", "sr 11-7 compliant", "sr 26-2 compliant", "certified",
    "the price is", "costs $", "priced at", "our price", "list price is",
    "$",  # any bare dollar figure in a guard say_this is an over-claim
)

_DEFAULT_EXPIRES_S = {"proactive": 45, "reactive": 300}

def _migrate_urgency(value: object) -> str:
    """Clamp urgency to two-value enum. Legacy "soon" -> "now" (fail-toward-seen)."""
    if isinstance(value, str):
        if value in CARD_URGENCIES:
            return value
        if value == "soon":
            return "now"
    return "fyi"


def _apply_say_this_gate(
    say_this: str | None,
    bullets: list[str],
    *,
    trigger: str | None,
    confidence: str,
    source: str,
    bullet_cap: int,
) -> tuple[str | None, list[str]]:
    """S4 say_this wall. GUARD triggers first (by construction), then the general
    confidence+source gate + medium 'Likely:' posture. Defensive: rewrites, never
    raises."""
    # HARD guard wall (bank over-claim safety): allowlist-only deferral shape.
    if trigger in GUARD_TRIGGERS:
        if say_this:
            low = say_this.lower()
            is_defer = any(m in low for m in _DEFER_MARKERS)
            is_overclaim = any(m in low for m in _OVERCLAIM_MARKERS)
            if not is_defer or is_overclaim:
                say_this = None  # drop anything that isn't a clean deferral
        return say_this, bullets  # no "Likely:" rewrite on guards

    grounded = source in SAY_THIS_SOURCES
    if confidence == "high" and grounded and say_this:
        return say_this, bullets

    dropped_for_confidence = bool(say_this) and confidence != "high"
    say_this = None
    if dropped_for_confidence and bullets:
        first = bullets[0]
        if not first.lower().startswith(_LIKELY_PREFIXES):
            bullets = [_clamp_str(f"Likely: {first}", bullet_cap)] + bullets[1:]
    return say_this, bullets


def parse_card(raw: object, *, lane: str) -> Card | None:
    """Defensive parse (unchanged contract: None only on non-dict / empty headline)."""
    if not isinstance(raw, dict):
        return None
    headline = _clamp_str(raw.get("headline") or "", MAX_HEADLINE_CHARS)
    if not headline:
        return None

    lane = lane if lane in CARD_LANES else "reactive"
    bullet_cap = MAX_WATCHER_BULLET_CHARS if lane == "proactive" else MAX_BULLET_CHARS
    card_type = _pick(raw.get("type"), CARD_TYPES, "heads_up")

    key_fact_raw = raw.get("key_fact")
    key_fact = (
        _clamp_str(key_fact_raw, MAX_KEY_FACT_CHARS) or None
        if key_fact_raw is not None else None
    )

    bullets_raw = raw.get("bullets")
    bullets: list[str] = []
    if isinstance(bullets_raw, list):
        for item in bullets_raw[:MAX_BULLETS]:
            text = _clamp_str(item, bullet_cap)
            if text:
                bullets.append(text)

    # cues: unchanged
    cues_raw = raw.get("cues")
    cues: list[str] = []
    if isinstance(cues_raw, list):
        for item in cues_raw[:MAX_CUES]:
            cue = " ".join(str(item).split()[:MAX_CUE_WORDS]).strip()
            if cue:
                cues.append(cue)

    say_this_raw = raw.get("say_this")
    say_this = str(say_this_raw).strip() or None if say_this_raw is not None else None
    trigger_raw = raw.get("trigger")
    trigger = str(trigger_raw).strip() or None if trigger_raw is not None else None

    confidence = _pick(raw.get("confidence"), CARD_CONFIDENCES, "medium")
    source = _pick(raw.get("source"), CARD_SOURCES, "transcript")
    urgency = _migrate_urgency(raw.get("urgency"))
    disposition = _pick(raw.get("disposition"), CARD_DISPOSITIONS, "render")
    update = bool(raw.get("update")) if isinstance(raw.get("update"), bool) else False

    say_this, bullets = _apply_say_this_gate(
        say_this, bullets, trigger=trigger, confidence=confidence,
        source=source, bullet_cap=bullet_cap,
    )

    # expires_in_s + topic_key: unchanged
    # ... (existing expires_in_s parse, existing topic_key slugify) ...

    return Card(
        id=f"card_{uuid.uuid4().hex[:12]}", lane=lane, type=card_type, trigger=trigger,
        headline=headline, key_fact=key_fact, bullets=bullets, cues=cues,
        say_this=say_this, confidence=confidence, urgency=urgency, source=source,
        disposition=disposition, update=update, expires_in_s=expires_in_s,
        topic_key=topic_key,
    )
```
